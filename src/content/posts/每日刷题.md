# 链表反转java代码
```
import java.util.*;
public class Solution {
    public ListNode ReverseList (ListNode head) {
        if(head==null){
            return head;
        }
        Stack<ListNode> stack= new Stack<>();
        //把节点全部入栈
        while(head!=null){
            stack.push(head);
            head = head.next;
        }

        ListNode node = stack.pop();
        ListNode dummy= node;
        while(stack.size()>0){
            node.next = stack.pop();
            node = node.next;
        }
        node.next=null;
        return dummy;
    }
}
```
